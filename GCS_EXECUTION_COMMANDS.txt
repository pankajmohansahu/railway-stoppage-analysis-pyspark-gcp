cat > main.py << 'EOFCF'
import logging
import csv
from io import StringIO
from google.cloud import storage

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def process_results(event, context):
    bucket_name = event['bucket']
    file_name = event['name']
    logger.info(f"Function triggered by upload of '{file_name}'")

    try:
        storage_client = storage.Client()
        bucket = storage_client.bucket(bucket_name)
        blob = bucket.blob(file_name)
        csv_content = blob.download_as_text(encoding='utf-8')

        csv_reader = csv.DictReader(StringIO(csv_content))
        max_avg = -1
        max_station = None

        for row in csv_reader:
            try:
                avg = float(row.get('avg_stop_duration', 0))
                if avg > max_avg:
                    max_avg = avg
                    max_station = row.get('Station Name', 'Unknown')
            except:
                continue

        logger.info("="*80)
        logger.info("SECTION C: CLOUD FUNCTION RESULTS")
        logger.info("="*80)
        logger.info(f"Station with HIGHEST AVERAGE STOP DURATION:")
        logger.info(f"  Station Name: {max_station}")
        logger.info(f"  Avg Stop Duration: {max_avg} minutes")
        logger.info("="*80)

        return f"Success: {max_station}"
    except Exception as e:
        logger.error(f"Error: {str(e)}")
        raise
EOFCF
